cmake_minimum_required(VERSION 3.22)
project(RNS8NativeDeviceConsumer LANGUAGES CXX)
find_package(RNS8 CONFIG REQUIRED)
set(RNS8_NATIVE_HIP_ROOT "" CACHE PATH "HIP SDK or ROCm root")
set(RNS8_NATIVE_GPU_TARGET "gfx1100" CACHE STRING "Explicit AMD GPU target")
set(RNS8_NATIVE_HIP_EXTRA_FLAGS "" CACHE STRING "Additional hipcc flags, such as an SDK STL compatibility include")
option(RNS8_NATIVE_USE_SHARED "Use the installed shared target" ON)
find_program(RNS8_NATIVE_HIPCC NAMES hipcc hipcc.bat HINTS "${RNS8_NATIVE_HIP_ROOT}/bin" REQUIRED)
find_path(RNS8_NATIVE_HIP_INCLUDE hip/hip_runtime_api.h HINTS "${RNS8_NATIVE_HIP_ROOT}/include" REQUIRED)
find_library(RNS8_NATIVE_HIP_LIBRARY NAMES amdhip64 HINTS "${RNS8_NATIVE_HIP_ROOT}/lib" "${RNS8_NATIVE_HIP_ROOT}/lib64" REQUIRED)
if(RNS8_NATIVE_USE_SHARED)
  if(NOT TARGET rns8::rns8)
    message(FATAL_ERROR "A shared RNS8 installation is required, or set RNS8_NATIVE_USE_SHARED=OFF")
  endif()
  set(_rns8 rns8::rns8)
else()
  set(_rns8 rns8::rns8_static)
endif()
set(_runtime)
if(MSVC)
  if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    list(APPEND _runtime -fms-runtime-lib=dll_dbg -D_DEBUG -D_ITERATOR_DEBUG_LEVEL=2)
  else()
    list(APPEND _runtime -fms-runtime-lib=dll -DNDEBUG -D_ITERATOR_DEBUG_LEVEL=0)
  endif()
endif()
set(_object "${CMAKE_CURRENT_BINARY_DIR}/producer_consumer${CMAKE_CXX_OUTPUT_EXTENSION}")
add_custom_command(OUTPUT "${_object}"
  COMMAND "${RNS8_NATIVE_HIPCC}" "--offload-arch=${RNS8_NATIVE_GPU_TARGET}" -std=c++17 -O2
    ${_runtime} ${RNS8_NATIVE_HIP_EXTRA_FLAGS} -c "${CMAKE_CURRENT_SOURCE_DIR}/producer_consumer.hip" -o "${_object}"
  DEPENDS producer_consumer.hip VERBATIM)
set_source_files_properties("${_object}" PROPERTIES GENERATED TRUE EXTERNAL_OBJECT TRUE)
add_executable(rns8-native-device-consumer main.cpp "${_object}")
target_compile_features(rns8-native-device-consumer PRIVATE cxx_std_17)
target_compile_definitions(rns8-native-device-consumer PRIVATE __HIP_PLATFORM_AMD__)
target_include_directories(rns8-native-device-consumer SYSTEM PRIVATE "${RNS8_NATIVE_HIP_INCLUDE}")
target_link_libraries(rns8-native-device-consumer PRIVATE ${_rns8} "${RNS8_NATIVE_HIP_LIBRARY}")
rns8_copy_runtime_dependencies(rns8-native-device-consumer)
if(WIN32 AND RNS8_NATIVE_USE_SHARED)
  add_custom_command(TARGET rns8-native-device-consumer POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:${_rns8}>" "$<TARGET_FILE_DIR:rns8-native-device-consumer>")
endif()
enable_testing()
add_test(NAME native-device-producer-consumer COMMAND rns8-native-device-consumer)
set_tests_properties(native-device-producer-consumer PROPERTIES TIMEOUT 120 LABELS "package;hip;interop")

add_test(NAME native-device-prepared-graph-pipeline COMMAND rns8-native-device-consumer --prepared)
set_tests_properties(native-device-prepared-graph-pipeline PROPERTIES TIMEOUT 120 LABELS "package;hip;interop;async;graph")
