cmake_minimum_required(VERSION 3.22)

project(RNS8DownstreamSmoke LANGUAGES CXX)

find_package(RNS8 CONFIG REQUIRED)
message(STATUS "Using RNS8 package: ${RNS8_DIR}")
option(RNS8_DOWNSTREAM_USE_SHARED "Consume the installed shared RNS8 target" OFF)
if(RNS8_DOWNSTREAM_USE_SHARED)
  if(NOT TARGET rns8::rns8)
    message(FATAL_ERROR "The installed RNS8 package has no shared target")
  endif()
  set(RNS8_DOWNSTREAM_TARGET rns8::rns8)
else()
  set(RNS8_DOWNSTREAM_TARGET rns8::rns8_static)
endif()

add_executable(rns8-downstream-smoke main.cpp)
add_executable(rns8-downstream-continuation exact_wide_cpu_continuation.cpp)
enable_testing()
foreach(consumer IN ITEMS rns8-downstream-smoke rns8-downstream-continuation)
  target_link_libraries(${consumer} PRIVATE ${RNS8_DOWNSTREAM_TARGET})
  target_compile_features(${consumer} PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${consumer} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(WIN32 AND RNS8_DOWNSTREAM_USE_SHARED)
    add_custom_command(TARGET ${consumer} POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "$<TARGET_FILE:${RNS8_DOWNSTREAM_TARGET}>" "$<TARGET_FILE_DIR:${consumer}>")
  endif()
  add_test(NAME ${consumer} COMMAND ${consumer})
  set_tests_properties(${consumer} PROPERTIES TIMEOUT 10 LABELS "package;cpu")
endforeach()
